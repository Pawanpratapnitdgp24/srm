SELECT
    SUM(T.PREMIUM_AM)                          AS TRANS_AM,
    P.POLICY_NO,
    TO_CHAR(P.INCEPTION_DT, 'yyyymmdd')        AS POLICY_EFF_DT,
    P.POLICY_TYPE_CD,
    C.DATE_SK                                  AS ACCOUNTING_DATE_SK   -- retain to match per month
FROM ttransaction T
JOIN tpolicy  P  ON T.policy_sk        = P.policy_sk
JOIN tdate    C  ON T.effective_dt_sk  = C.date_sk        -- effective date dimension (CIP/DC/Instanda pattern)
JOIN tdate    T3 ON T.transaction_dt_sk = T3.date_sk       -- transaction date dimension (US536949 back-dated logic)
WHERE
    P.POLICY_NO IN (
        '2500000433','2500000413','9394501580','4500001881','3620000723',
        '5390003513','5390004108','5390004110','2500000430','9394501690',
        '4500001853','3620000716','5390004129','0063100349','2500000426',
        '0025085481','5390004309','5390003696','5390003699','2500000448'
    )
    AND P.SOURCE_SYSTEM_SK IN (16, 17, 18, 19, 20, 21)     -- all CIP/DC/Instanda systems
    AND T.PREMIUM_AM <> 0                                   -- exclude zero-premium rows
    AND (
        -- Normal: effective month = accounting month, transaction posted in same or earlier month
        (C.DATE_SK IN (9405, 9436) AND T3.DATE_SK <= C.DATE_SK)
        OR
        -- Back-dated: effective month is prior, but transaction posted in the current month
        (C.DATE_SK NOT IN (9405, 9436) AND T3.DATE_SK IN (9405, 9436))
    )
GROUP BY
    P.POLICY_NO,
    TO_CHAR(P.INCEPTION_DT, 'yyyymmdd'),
    P.POLICY_TYPE_CD,
    C.DATE_SK;



SELECT
    A.ACCOUNTING_DT,
    A.TRAN_EFF_DT,
    A.POLICY_NO,
    A.ACCOUNTING_MONTH,
    A.PREMIUM_LOSS_IND,
    A.SRM_AMOUNT,
    D.TRANS_AM,
    D.POLICY_TYPE_CD,
    NVL(A.SRM_AMOUNT, 0) - NVL(D.TRANS_AM, 0) AS AMOUNT_DIFFERENCE
FROM (
    -- ── SRM side: unchanged ──────────────────────────────────────────────
    SELECT
        ACCOUNTING_DT,
        TRAN_EFF_DT,
        POLICY_NO,
        ACCOUNTING_MONTH,
        PREMIUM_LOSS_IND,
        SUM(TRAN_AM) AS SRM_AMOUNT
    FROM STAGE_MISREG.STAGE_SRM_PREMIUM_LOSS
    WHERE POLICY_NO IN (
        '2500000433','2500000413','9394501580','4500001881','3620000723',
        '5390003513','5390004108','5390004110','2500000430','9394501690',
        '4500001853','3620000716','5390004129','0063100349','2500000426',
        '0025085481','5390004309','5390003696','5390003699','2500000448'
    )
    GROUP BY
        ACCOUNTING_DT, TRAN_EFF_DT, POLICY_NO, ACCOUNTING_MONTH, PREMIUM_LOSS_IND
) A

FULL OUTER JOIN (
    /*
     * FIX 1+2: Replace calendar_month_sk with the two-date pcs_month_acctng_ct
     *          logic that prc_premium_feed_load actually uses (US536949).
     *          t_eff  → effective_dt_sk  → pcs_month_acctng_ct
     *          t_tran → transaction_dt_sk → pcs_month_acctng_ct
     *
     * FIX 3: tran_eff_dt is TO_CHAR(t_eff.actual_dt) — the transaction's
     *        effective date, NOT tpolicy.inception_dt.
     *
     * FIX 4: accounting_month added to GROUP BY so each month is a separate
     *        row, matching the granularity of stage_srm_premium_loss.
     *
     * FIX 5: source_system_sk on tpolicy, range (16-21).
     *
     * FIX 6: TAX_FEE_SRCHRG_AM = 0 filter removed.
     *        Instanda SRM uses raw premium_am (the tax deduction is
     *        commented out in prc_premium_feed_load).
     *        For HO/CHO: all rows allowed (HAVING filters zero-sum groups).
     *        For others:  WHERE premium_am <> 0.
     *
     * FIX 7: COVERAGE_SK <> '0' filter removed — not in SRM code.
     *
     * FIX 8: Policy type filter added to match SRM's two blocks.
     *
     * FIX 9: HAVING clause added for HO/CHO zero-sum group exclusion.
     */
    SELECT
        t_eff.pcs_month_acctng_ct                  AS ACCOUNTING_MONTH,
        TO_CHAR(t_eff.actual_dt, 'yyyymmdd')        AS TRAN_EFF_DT,   -- FIX 3
        P.POLICY_NO,
        P.POLICY_TYPE_CD,
        SUM(T.PREMIUM_AM)                           AS TRANS_AM       -- FIX 6: raw, no tax deduction
    FROM ttransaction T
    JOIN tpolicy      P    ON T.policy_sk          = P.policy_sk
    -- FIX 1: join on effective_dt_sk (not calendar_month_sk)
    JOIN tdate        t_eff  ON T.effective_dt_sk  = t_eff.date_sk
    -- FIX 1: second date join for transaction_dt_sk
    JOIN tdate        t_tran ON T.transaction_dt_sk = t_tran.date_sk
    WHERE P.POLICY_NO IN (
        '2500000433','2500000413','9394501580','4500001881','3620000723',
        '5390003513','5390004108','5390004110','2500000430','9394501690',
        '4500001853','3620000716','5390004129','0063100349','2500000426',
        '0025085481','5390004309','5390003696','5390003699','2500000448'
    )
    -- FIX 5: source_system_sk from tpolicy, full range
    AND P.source_system_sk IN (16, 17, 18, 19, 20, 21)
    -- FIX 8: restrict to SRM-processed policy types only
    AND P.POLICY_TYPE_CD IN ('HO','CHO','AU','PE','EXFL','PC','EQ','CAU','YHT')
    -- FIX 2: pcs_month_acctng_ct backdated-transaction logic (US536949)
    AND (
        (t_eff.pcs_month_acctng_ct  = t_tran.pcs_month_acctng_ct   -- same-month normal
             AND t_tran.pcs_month_acctng_ct <= t_eff.pcs_month_acctng_ct)
        OR (t_eff.pcs_month_acctng_ct < t_tran.pcs_month_acctng_ct) -- backdated
    )
    -- FIX 6: HO/CHO gets ALL rows (HAVING handles zero groups);
    --        all other types need premium_am <> 0
    AND (
        P.POLICY_TYPE_CD IN ('HO', 'CHO')
        OR T.PREMIUM_AM <> 0
    )
    -- FIX 4: group includes ACCOUNTING_MONTH and FIX 3: TRAN_EFF_DT
    GROUP BY
        t_eff.pcs_month_acctng_ct,
        TO_CHAR(t_eff.actual_dt, 'yyyymmdd'),
        P.POLICY_NO,
        P.POLICY_TYPE_CD
    -- FIX 9: for HO/CHO exclude zero-sum groups; for others pass everything
    HAVING
        (P.POLICY_TYPE_CD IN ('HO','CHO') AND SUM(T.PREMIUM_AM) <> 0)
        OR P.POLICY_TYPE_CD NOT IN ('HO','CHO')
) D
ON  A.POLICY_NO        = D.POLICY_NO
AND A.TRAN_EFF_DT      = D.TRAN_EFF_DT        -- FIX 3: both now = transaction eff date
AND A.ACCOUNTING_MONTH = D.ACCOUNTING_MONTH   -- FIX 4: month-level match
WHERE NVL(A.SRM_AMOUNT, 0) <> NVL(D.TRANS_AM, 0)
ORDER BY
    COALESCE(A.POLICY_NO,        D.POLICY_NO),
    COALESCE(A.TRAN_EFF_DT,      D.TRAN_EFF_DT),
    COALESCE(A.ACCOUNTING_MONTH, D.ACCOUNTING_MONTH);
